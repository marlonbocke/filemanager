function getAllowedFilePath(string $inputPath): ?string
{
    $root = realpath(__DIR__ . '/../private/files');
    $path = realpath($inputPath);

    if ($root === false || $path === false) {
        return null;
    }

    $allowedPrefix = rtrim($root, DIRECTORY_SEPARATOR) . DIRECTORY_SEPARATOR;

    if (!str_starts_with($path, $allowedPrefix)) {
        return null;
    }

    return $path;
}
